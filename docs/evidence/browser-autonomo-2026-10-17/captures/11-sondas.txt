import { createRequire } from 'node:module';
import { readFileSync, writeFileSync } from 'node:fs';
const REPO = '/home/douglas/WebstormProjects/preco-que-da-lucro';
const require = createRequire(`${REPO}/package.json`);
const { chromium } = require('playwright-core');
const BASE = 'http://127.0.0.1:4173';
const creds = {};
for (const l of readFileSync(`${REPO}/.artifacts/e2e-local.env`, 'utf8').split('\n')) { const i = l.indexOf('='); if (i > 0) creds[l.slice(0, i)] = l.slice(i + 1); }

const browser = await chromium.launch({ channel: 'chrome', headless: true });

const page = await ctx.newPage();
const log = {};
const ck = async () => (await ctx.cookies([BASE])).map((c) => ({ name: c.name, domain: c.domain, path: c.path, secure: c.secure, httpOnly: c.httpOnly, sameSite: c.sameSite, expires: c.expires }));

await page.goto(`${BASE}/auth`, { waitUntil: 'domcontentloaded' });
await page.getByLabel(/e-?mail/i).first().fill(creds.E2E_AUTH_EMAIL);
await page.getByLabel(/senha/i).first().fill(creds.E2E_AUTH_PASSWORD);
await page.getByRole('button', { name: /entrar|acessar|continuar com e-?mail/i }).first().click();
await page.waitForTimeout(2500);
log.urlAposLogin = page.url();
log.cookiesAposLogin = await ck();

// a sessao vale para o servidor?
const s1 = await page.request.get(`${BASE}/api/auth/get-session`);
log.getSessionAposLogin = s1.status();
log.getSessionCorpo = (await s1.text().catch(() => '')).slice(0, 80);

// e depois de um reload?
await page.goto(`${BASE}/inicio`, { waitUntil: 'domcontentloaded' });
await page.waitForTimeout(1500);
log.urlAposInicio = page.url();
log.cookiesAposInicio = await ck();
const s2 = await page.request.get(`${BASE}/api/auth/get-session`);
log.getSessionAposInicio = s2.status();

// e sem cookie nenhum, o proprio /inicio responde o que?
const s3 = await page.request.get(`${BASE}/inicio`, { maxRedirects: 0 });
log.inicioSemCookies = s3.status();

await ctx.storageState({ path: `${REPO}/.artifacts/cu/probe-storage.json` });
const st = JSON.parse(readFileSync(`${REPO}/.artifacts/cu/probe-storage.json`, 'utf8'));
log.storageState = { cookies: st.cookies.map((c) => c.name), origins: st.origins.length };

console.log(JSON.stringify(log, null, 1));
await browser.close();

import { readFileSync } from 'node:fs';

const creds = {};
for (const l of readFileSync(`${REPO}/.artifacts/e2e-local.env`, 'utf8').split('\n')) { const i = l.indexOf('='); if (i > 0) creds[l.slice(0, i)] = l.slice(i + 1); }

const browser = await chromium.launch({ channel: 'chrome', headless: true });
const ctx = await browser.newContext({ baseURL: BASE, extraHTTPHeaders: { 'x-forwarded-for': '198.51.100.10' } });

const estado = async (rotulo) => {
  const cookies = (await ctx.cookies([BASE])).map((c) => c.name);
  const el = await page.locator('body').innerText().catch(() => '');
  console.log(JSON.stringify({ rotulo, url: new URL(page.url()).pathname, cookies, emailVisivel: el.includes('teste@example.test'), h1: (await page.locator('h1, h2').first().textContent().catch(() => null))?.trim().slice(0, 40) }));

// login pela API, como o driver faz
const api = await request.newContext({ baseURL: BASE, extraHTTPHeaders: { origin: BASE, 'x-forwarded-for': '198.51.100.10' } });

console.log('API sign-in:', res.status(), '| cookies:', st.cookies.map((c) => `${c.name}(secure=${c.secure})`).join(','));
