#!/usr/bin/env bash
# Selo do WP-R4 (manifesto gerado/conferido pela ferramenta compartilhada).
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts; SELO=docs/evidence/seal-kpi-item16-2026-09-20; CAP=$SELO/captures; FAIL=0
echo "=== selo do WP-R4 ==="
for f in r4-s6red.log r4-s6green.log r4-gate.log r4-acceptance.log r4-s7.log adversarial-r4-verdict.md; do
  [ -f "$A/$f" ] || { echo "  !! FALTA: $A/$f"; FAIL=1; }
done
[ "$FAIL" -eq 0 ] || { echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; exit 1; }
GENV=$(grep -c "CHECK_EXIT=0" "$A/r4-gate.log" || true)
[ "$GENV" -ge 1 ] || { echo "  !! FALHA: gate sem CHECK_EXIT=0"; exit 1; }
rm -rf "$CAP"; mkdir -p "$CAP"
cp "$A/r4-s6red.log" "$CAP/s6-red-tests.log.txt"
cp "$A/r4-s6green.log" "$CAP/s6-green-tests.log.txt"
cp "$A/r4-gate.log" "$CAP/gate-local.log.txt"
cp "$A/r4-acceptance.log" "$CAP/acceptance.log.txt"
cp "$A/r4-s7.log" "$CAP/s7-guard.log.txt"
cp "$A/adversarial-r4-verdict.md" "$CAP/adversarial-r4-verdict.md.txt"
cp scripts/m02-seal.mjs "$CAP/seal.mjs.txt"
cp scripts/m02-work-package-guard.mjs "$CAP/guard.mjs.txt"
cp scripts/lib/m02-ci-coverage.ts "$CAP/m02-ci-coverage.ts.txt"
cp scripts/lib/bundle-identity.mjs "$CAP/bundle-identity.mjs.txt"
cp scripts/check-bundle.mjs "$CAP/check-bundle.mjs.txt"
cp docs/evidence/_templates/work-package.md "$CAP/work-package.md.txt"
cp "$A/zz-r4-s7.sh" "$CAP/s7-guard.sh.txt"
cp "$A/zz-r4-seal.sh" "$CAP/selo.sh.txt"
: > "$CAP/hashes.txt"
for f in scripts/m02-seal.mjs scripts/m02-seal.d.mts scripts/lib/m02-ci-coverage.ts \
         scripts/lib/bundle-identity.mjs scripts/lib/bundle-identity.d.mts scripts/check-bundle.mjs \
         scripts/m02-work-package-guard.mjs docs/evidence/_templates/work-package.md AGENTS.md; do
  if [ -f "$f" ]; then sha256sum "$f" >> "$CAP/hashes.txt"; else echo "  !! FALTA no hashes: $f"; FAIL=1; fi
done
npx prettier --write "$SELO/README.md" "$SELO/SPEC.md" >/dev/null 2>&1
npx prettier --check "$SELO/README.md" "$SELO/SPEC.md" > "$A/r4-prettier.log" 2>&1 || { echo "  !! FALHA: prettier"; FAIL=1; }
echo "--- manifesto via scripts/m02-seal.mjs (dogfooding) ---"
node scripts/m02-seal.mjs --dir "$SELO" --write || FAIL=1
node scripts/m02-seal.mjs --dir "$SELO" || FAIL=1
if [ "$FAIL" -eq 0 ]; then echo "SELO: OK"; echo "SEAL_SCRIPT_EXIT=0"; else echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; fi
