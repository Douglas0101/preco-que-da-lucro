  analysisId: "fixture-current-main-analysis",
  ceTaskId: "fixture-main-ce-task",
  providerIdentity: { revision: mainSha, branch: "main", pullRequest: null },
};
function cli(name: string, readout?: object, failure?: string) {
  const dir = mkdtempSync(resolve(root, "baseline-bench-"));
  try {
    if (readout) writeFileSync(resolve(dir, "c25-gate-readout.json"), JSON.stringify(readout));
    const result = spawnSync(process.execPath, [resolve(cwd, "scripts/sonar/main-baseline.mjs")], {
      cwd,
      encoding: "utf8",
      env: {
        PATH: process.env.PATH,
        HOME: dir,
        RUNNER_TEMP: dir,
        GITHUB_TOKEN: "fixture-only-not-a-credential",
        SONAR_TOKEN: "fixture-only-not-a-credential",
        NODE_OPTIONS: `--import=${resolve(root, "baseline-api-fixture.mjs")}`,
      },
    });
    const report = JSON.parse(readFileSync(resolve(dir, "c24-main-baseline.json"), "utf8"));
    assert.equal(result.status, failure ? 2 : 0, result.stderr);
    if (failure) {
      assert.equal(report.reasonCode, failure);
      assert.equal(report.verdict, "NO-VERDICT");
      assert.equal(report.requests.length, 1);
      assert.equal(report.requests[0].origin, "github");
      assert.equal(report.requests[0].status, 200);
    } else {
      assert.equal(report.mainSha, mainSha);
      assert.equal(report.analysisId, mainReadout.analysisId);
      assert.equal(report.metrics.new_coverage, 80);
      assert.equal(report.requests.length, 6);
      assert.ok(report.period);
    }
    console.log(JSON.stringify({ scenario: name, exit: result.status, reasonCode: report.reasonCode ?? null, mainSha: report.mainSha ?? null, analysisId: report.analysisId ?? null, requests: report.requests.map(({ origin, path, status }: { origin: string; path: string; status: number }) => ({ origin, path, status })) }));
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }
}
