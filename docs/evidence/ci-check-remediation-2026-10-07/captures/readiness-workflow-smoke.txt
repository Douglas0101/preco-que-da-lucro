  ["direct", direct],
  ["pooled", pooled],

  generatedAt: new Date().toISOString(),
  direct: { connected: false },
  pooled: { connected: false },
};

for (const [kind, connectionString] of connections) {
  const client = new Client({ connectionString });
  try {
    await client.connect();
    const result = await client.query(
      "select current_database() as database, current_user as user, current_setting('server_version_num')::int as version_num",

    const postgresMajor = Math.floor(Number(result.rows[0]?.version_num) / 10000);
    if (postgresMajor !== expectedMajor) {

    evidence[kind] = {

      database: result.rows[0]?.database ?? null,
      user: result.rows[0]?.user ?? null,
      postgresMajor,
    };
    console.log(`${kind}: connected on PostgreSQL ${postgresMajor}`);

    await client.end();
