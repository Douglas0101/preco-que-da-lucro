#!/usr/bin/env bash
# Selo do WP-R3.
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts; SELO=docs/evidence/wp-r3-forensics-2026-09-20; CAP=$SELO/captures; FAIL=0
echo "=== selo do WP-R3 ==="
for f in r3-gate.log r3-s7.log adversarial-r3-verdict.md; do
  [ -f "$A/$f" ] || { echo "  !! FALTA: $A/$f"; FAIL=1; }
done
[ "$FAIL" -eq 0 ] || { echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; exit 1; }
GENV=$(grep -c "CHECK_EXIT=0" "$A/r3-gate.log" || true)
[ "$GENV" -ge 1 ] || { echo "  !! FALHA: gate sem CHECK_EXIT=0"; exit 1; }
rm -rf "$CAP"; mkdir -p "$CAP"
cp "$A/r3-gate.log" "$CAP/gate-local.log.txt"
cp "$A/r3-s7.log" "$CAP/s7-guard.log.txt"
cp "$A/zz-r3-s7.sh" "$CAP/s7-guard.sh.txt"
cp "$A/r3-s7-diff.txt" "$CAP/s7-diff.txt"
cp "$A/adversarial-r3-verdict.md" "$CAP/adversarial-r3-verdict.md.txt"
cp "$A/ci-binding.log" "$CAP/ci-binding.log.txt"
cp "$A/guard-falsifica.log" "$CAP/guard-falsifica.log.txt"
cp "$A/calib-a.md" "$CAP/calib-a.md.txt"
cp "$A/calib-b1.md" "$CAP/calib-b1.md.txt"
cp "$A/calib-b2.md" "$CAP/calib-b2.md.txt"
cp scripts/m02-work-package-guard.mjs "$CAP/guard.mjs.txt"
cp docs/evidence/_templates/work-package.md "$CAP/work-package.md.txt"
cp docs/evidence/analise-avancada-fase0-wpR2-wpR0-2026-09-20.md "$CAP/analise-transcrita.md.txt"
cp "$A/zz-r3-seal.sh" "$CAP/selo.sh.txt"
cp "$A/template-sem-item15.md"    "$CAP/scratch-item15-removido.md.txt"
cp "$A/template-origem-na.md"    "$CAP/scratch-origem-na.md.txt"
cp "$A/template-coluna-extra.md" "$CAP/scratch-coluna-extra.md.txt"
# gate de calibracao: LINHA DE VEREDICTO por arquivo (nao a palavra na prosa)
RA=$(grep -cF -- '**C1 — REJECTED**' "$CAP/calib-a.md.txt" || true)
RB=$(grep -cF -- '**C1 — REJECTED**' "$CAP/calib-b1.md.txt" || true)
UB=$(grep -cF -- '**C1 — UNVERIFIABLE**' "$CAP/calib-b2.md.txt" || true)
echo "  calibracao por linha de veredicto: A=$RA B1=$RB (esperado 1) · B2=$UB (esperado 1)"
[ "$RA" -eq 1 ] && [ "$RB" -eq 1 ] && [ "$UB" -eq 1 ] || { echo "  !! FALHA: calibracao nao cobre as duas categorias"; FAIL=1; }
: > "$CAP/hashes.txt"; OK=1
for f in docs/evidence/_templates/work-package.md AGENTS.md scripts/m02-work-package-guard.mjs \
         package.json .github/workflows/ci-light.yml \
         docs/evidence/analise-avancada-fase0-wpR2-wpR0-2026-09-20.md \
         docs/evidence/wp-r3-forensics-2026-09-20/SPEC.md; do
  if [ -f "$f" ]; then sha256sum "$f" >> "$CAP/hashes.txt"; else echo "  !! FALTA no hashes: $f"; OK=0; fi
done
[ "$OK" -eq 1 ] || FAIL=1
npx prettier --write "$SELO/README.md" "$SELO/SPEC.md" >/dev/null 2>&1
npx prettier --check "$SELO/README.md" "$SELO/SPEC.md" > "$A/r3-prettier.log" 2>&1 || { echo "  !! FALHA: prettier"; FAIL=1; }
rm -f "$SELO/MANIFEST.sha256"
find "$SELO" -type f ! -name 'MANIFEST.sha256' -print0 | sort -z | xargs -0 sha256sum > "$SELO/MANIFEST.sha256"
D=$(find "$SELO" -type f ! -name 'MANIFEST.sha256' | wc -l); L=$(wc -l < "$SELO/MANIFEST.sha256")
echo "  descobertos=$D listados=$L"; [ "$D" -eq "$L" ] || { echo "  !! cardinalidade"; FAIL=1; }
LC_ALL=C sha256sum -c "$SELO/MANIFEST.sha256" > "$A/r3-manifest-check.txt" 2>&1
N=$(grep -c ': FAILED$' "$A/r3-manifest-check.txt" || true); echo "  NAO-OK=$N"; [ "$N" -eq 0 ] || FAIL=1
if [ "$FAIL" -eq 0 ]; then echo "SELO: OK"; echo "SEAL_SCRIPT_EXIT=0"; else echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; fi
