function vitestJson(target, tag) {
  const out = ART + "/s6-sum-" + tag + ".json";
  const r = spawnSync(process.execPath, ["node_modules/.bin/vitest", "run", target, "--config", ".artifacts/s6-vitest3.config.ts", "--reporter=json", "--outputFile.json=" + out], { cwd: REPO, encoding: "utf8", maxBuffer: 100 * 1024 * 1024 });
  const j = JSON.parse(readFileSync(out, "utf8"));
  const keys = ["numTotalTests", "numPassedTests", "numFailedTests", "numPendingTests", "numTodoTests", "numTotalTestSuites", "success"];
  console.log("--- vitest json for " + target + " (exit " + String(r.status) + ") ---");
  for (const k of keys) console.log("    " + k + " = " + JSON.stringify(j[k]) + "  (typeof " + typeof j[k] + ")");
  return { j, status: r.status };
}

function build(name, target) {
  let out = src;
  const reps = [
    ['const testFile = "src/test/product-contracts.test.ts";', 'const testFile = "' + target + '";'],
    ['const repoRoot = fileURLToPath(new URL("../..", import.meta.url));', 'const repoRoot = "' + REPO + '";'],
    ['import { requireAdminUrl } from "./migrate";', 'import { requireAdminUrl } from "../scripts/db/migrate";'],
    ['"--reporter=default",', '"--config", ".artifacts/s6-vitest3.config.ts", "--reporter=default",'],
  ];
  for (const r of reps) out = out.replace(r[0], r[1]);
  if (out === src) throw new Error("no patch applied");
  writeFileSync(ART + "/" + name, out);
  return ART + "/" + name;
}
function run(label, script, admin) {
  const env = { ...process.env };
  env.DATABASE_ADMIN_URL = admin;
  env.DATABASE_URL = admin;
  env.DATABASE_DRIVER = "node-postgres";
  delete env.DATABASE_URL_UNPOOLED;
  const r = spawnSync(process.execPath, ["node_modules/.bin/tsx", script], { cwd: REPO, env, encoding: "utf8", maxBuffer: 100 * 1024 * 1024 });
  const lines = ((r.stdout || "") + (r.stderr || "")).split("\n").map((x) => x.trim()).filter(Boolean);
  console.log("### " + label + "  exit=" + String(r.status));
  for (const x of lines.filter((l) => /Tests |AssertionError|produziu|passed \(|incompleta|pulada/.test(l)).slice(-3)) console.log("    | " + x);
  console.log("");
}
