#!/usr/bin/env python3
"""Colhe o veredicto adversarial do WP2 da lane do bg_delegate e confere o sha256.

O sha256 que a lane declara e' o dos BYTES DECODIFICADOS do campo `answer`.
Conferir depois de acrescentar newline da' divergencia falsa (bug medido no WP1).
"""
import base64
import hashlib
import json
import pathlib
import sys

RAIZ = pathlib.Path(__file__).resolve().parents[1]

TASK_ID = "d1d251651a88776bdaf3c9e70d390684a"
# A lane do delegate vive no .pi/ da sessao-mae (o repo principal), nao no worktree.
_LANE_REL = pathlib.Path(".pi") / "delegate" / "01a0ba68-f0bb-7334-a6ef-44c1a7da4485-163167" / TASK_ID
_CANDIDATOS = [
    RAIZ / ".." / _LANE_REL,
    RAIZ / _LANE_REL,
]
LANE = next((c.resolve() for c in _CANDIDATOS if (c / "result.json").exists()), None)
if LANE is None:
    print("COLHE_EXIT=2 LANE NAO ENCONTRADA em:")
    for c in _CANDIDATOS:
        print(f"  {c.resolve()}")
    sys.exit(2)

RESULT = LANE / "result.json"
DEST = RAIZ / "docs" / "evidence" / "d1-gate-ux-2026-09-19" / "captures" / "adversarial-wp2-verdict.md.txt"

if not RESULT.exists():
    print(f"COLHE_EXIT=2 RESULT AUSENTE: {RESULT}")
    sys.exit(2)

dados = json.loads(RESULT.read_text())
answer = dados.get("answer")
if not isinstance(answer, dict):
    print(f"COLHE_EXIT=2 answer nao e' dict: {type(answer)}")
    sys.exit(2)

blocos = answer.get("blocks") or []
partes = []
for bloco in blocos:
    data = bloco.get("data_base64")
    if data:
        partes.append(base64.b64decode(data))
bytes_decodificados = b"".join(partes)

sha_declarado = answer.get("sha256")
sha_medido = hashlib.sha256(bytes_decodificados).hexdigest()

print(f"bytes decodificados: {len(bytes_decodificados)}")
print(f"sha256 declarado pela lane: {sha_declarado}")
print(f"sha256 dos bytes medido:    {sha_medido}")

DEST.parent.mkdir(parents=True, exist_ok=True)
DEST.write_bytes(bytes_decodificados + b"\n")
tamanho = DEST.stat().st_size
print(f"gravado: {DEST.relative_to(RAIZ)} ({tamanho} B com a newline final)")

if sha_declarado == sha_medido:
    print("CADEIA DE CUSTODIA: OK")
    print("COLHE_EXIT=0")
    sys.exit(0)

print("CADEIA DE CUSTODIA: DIVERGE")
print("COLHE_EXIT=2")
sys.exit(2)
