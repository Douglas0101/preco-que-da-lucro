#!/usr/bin/env bash
# Selo do WP4 (F-B-mem-purge). Convencao: captures/ + *.txt (nunca raw/, nunca *.log).
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts
SELO=docs/evidence/f-b-mem-purge-2026-09-20
CAP=$SELO/captures
FAIL=0

echo "=== selo do WP4 ==="
echo "  HEAD: $(git rev-parse HEAD)"

echo "=== 0. capturas obrigatorias presentes ==="
for f in wp4-probe-red.log wp4-probe-green.log wp4-s7.log wp4-gate.log wp4-db-test.log wp4-db-check.log \
         zz-wp4-probe.sh zz-wp4-s7.sh adversarial-wp4-verdict.md; do
  if [ ! -f "$A/$f" ]; then echo "  !! FALTA: $A/$f"; FAIL=1; fi
done
[ "$FAIL" -eq 0 ] || { echo "SELO: REPROVADO (captura ausente)"; echo "SEAL_SCRIPT_EXIT=1"; exit 1; }

echo "=== 1. captures do zero (o selo nao herda restos) ==="
rm -rf "$CAP"; mkdir -p "$CAP"

copiar() { cp "$A/$1" "$CAP/$2"; }
copiar wp4-probe-red.log          probe-red.log.txt
copiar wp4-probe-green.log        probe-green.log.txt
copiar zz-wp4-probe.sh               probe.sh.txt
for nome in seed-auth explain backfill auth-integ ai-budget; do
  copiar "wp4-red-$nome.log"      "red-$nome.log.txt"
  copiar "wp4-green-$nome.log"    "green-$nome.log.txt"
done
copiar wp4-migrate.log            scaffold-migrate.log.txt
copiar wp4-trigger.log            scaffold-trigger.log.txt
copiar wp4-preseed.log            scaffold-preseed.log.txt
copiar wp4-typecheck.log          typecheck.log.txt
copiar wp4-lint.log               lint.log.txt
copiar wp4-gate.log               gate-local.log.txt
copiar wp4-db-test.log            db-test.log.txt
copiar wp4-db-check.log           db-check.log.txt
copiar wp4-s7.log                 s7-guard.log.txt
copiar zz-wp4-s7.sh               s7-guard.sh.txt
copiar wp4-s7-diff.txt            s7-diff.txt
copiar adversarial-wp4-verdict.md adversarial-wp4-verdict.md.txt
copiar wp4-docker-id.txt          docker-id.txt
for f in scripts/db/purge-fixtures.ts scripts/e2e/seed-auth.ts scripts/db/explain-evidence.ts \
         scripts/db/test-backfill.ts scripts/db/test-auth-integration.ts scripts/db/test-ai-budget.ts; do
  cp "$f" "$CAP/$(basename "$f").txt"
done
cp "$A/zz-wp4-seal.sh" "$CAP/selo.sh.txt"
: > "$CAP/hashes.txt"
HASHES_OK=1
for f in scripts/db/purge-fixtures.ts scripts/e2e/seed-auth.ts scripts/db/explain-evidence.ts \
         scripts/db/test-backfill.ts scripts/db/test-auth-integration.ts scripts/db/test-ai-budget.ts \
         .artifacts/zz-wp4-probe.sh .artifacts/wp4-probe-red.log .artifacts/wp4-probe-green.log; do
  if [ -f "$f" ]; then
    sha256sum "$f" >> "$CAP/hashes.txt"
  else
    echo "  !! FALTA no hashes: $f"
    HASHES_OK=0
  fi
done
[ "$HASHES_OK" -eq 1 ] || { echo "  !! FALHA: hashes com arquivo ausente"; FAIL=1; }
echo "  arquivos em captures/: $(ls -1 "$CAP" | wc -l)"

echo "=== 2. armadilhas do .gitignore ==="
echo "  arquivos em raw/: $(find "$SELO" -path '*/raw/*' -type f | wc -l) (esperado 0)"
echo "  arquivos *.log (nao .log.txt): $(find "$SELO" -name '*.log' -type f | wc -l) (esperado 0)"
IGNORADOS=$(find "$SELO" -type f -print0 | xargs -0 git check-ignore 2>/dev/null | wc -l)
echo "  arquivos do selo ignorados pelo git: $IGNORADOS (esperado 0)"
[ "$IGNORADOS" -eq 0 ] || { echo "  !! FALHA: arquivo do selo seria ignorado"; FAIL=1; }

echo "=== 3. prettier nos .md do selo ==="
npx prettier --write "$SELO/README.md" "$SELO/SPEC.md" > /dev/null 2>&1
npx prettier --check "$SELO/README.md" "$SELO/SPEC.md" > "$A/wp4-prettier.log" 2>&1
PRETTIER=$?
echo "  PRETTIER_CHECK_EXIT=$PRETTIER"
[ "$PRETTIER" -eq 0 ] || { echo "  !! FALHA: prettier"; FAIL=1; }

echo "=== 4. manifesto (da raiz do repo) ==="
rm -f "$SELO/MANIFEST.sha256"
find "$SELO" -type f ! -name 'MANIFEST.sha256' -print0 | sort -z | xargs -0 sha256sum > "$SELO/MANIFEST.sha256"
FORA=$(grep -cvE "^[0-9a-f]{64}  $SELO/" "$SELO/MANIFEST.sha256" || true)
echo "  caminhos fora do selo no manifesto: $FORA (esperado 0)"
[ "$FORA" -eq 0 ] || { echo "  !! FALHA: manifesto com caminho fora do selo"; FAIL=1; }
DESCOBERTOS=$(find "$SELO" -type f ! -name 'MANIFEST.sha256' | wc -l)
LISTADOS=$(wc -l < "$SELO/MANIFEST.sha256")
echo "  descobertos=$DESCOBERTOS listados=$LISTADOS"
[ "$DESCOBERTOS" -eq "$LISTADOS" ] && echo "  checked === discovered: OK" || { echo "  !! FALHA: cardinalidade"; FAIL=1; }

echo "=== 5. verificacao do manifesto ==="
LC_ALL=C sha256sum -c "$SELO/MANIFEST.sha256" > "$A/wp4-manifest-check.txt" 2>&1
OK=$(grep -c ': OK$' "$A/wp4-manifest-check.txt" || true)
NAOOK=$(grep -c ': FAILED$' "$A/wp4-manifest-check.txt" || true)
echo "  OK=$OK NAO-OK=$NAOOK"
[ "$NAOOK" -eq 0 ] || { echo "  !! FALHA: manifesto nao confere"; FAIL=1; }

echo
echo "  tamanho do selo: $(du -sh "$SELO" | cut -f1)"
if [ "$FAIL" -eq 0 ]; then echo "SELO: OK"; echo "SEAL_SCRIPT_EXIT=0"; else echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; fi
