#!/usr/bin/env bash
# Selo do WP3 (F-D2-depth-pin). Convencao captures/ + *.txt (nunca raw/, nunca *.log).
# Licao do WP2: o selo e funcao do script, nao do historico do diretorio.
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts
SELO=docs/evidence/d2-depth-pin-2026-09-20
CAP=$SELO/captures
FAIL=0

echo "=== selo do WP3 ==="
echo "  HEAD: $(git rev-parse HEAD)"

echo "=== 1. captures do zero (o selo nao herda restos) ==="
rm -rf "$CAP"; mkdir -p "$CAP"

cp "$A/wp3-falsifica.log"        "$CAP/falsifica.log.txt"
cp "$A/zz-wp3-falsifica.sh"      "$CAP/falsifica.sh.txt"
cp "$A/wp3-verify.log"           "$CAP/bateria.log.txt"
cp "$A/zz-wp3-verify.sh"         "$CAP/bateria.sh.txt"
cp "$A/wp3-gate.log"             "$CAP/gate-local.log.txt"
cp "$A/wp3-s7.log"               "$CAP/s7-guard.log.txt"
cp "$A/zz-wp3-s7.sh"             "$CAP/s7-guard.sh.txt"
cp "$A/wp3-s7-diff.txt"          "$CAP/s7-diff.txt"
cp "$A/wp3-runner-piso.txt"      "$CAP/runner-piso.log.txt"
cp src/test/products-fk-conflict.test.ts      "$CAP/suite-products-fk-conflict.ts.txt"
cp scripts/db/test-products-fk-conflict.ts    "$CAP/runner-products-fk-conflict.ts.txt"
cp src/lib/products.functions.ts              "$CAP/runtime-products.functions.ts.txt"
cp "$A/zz-wp3-seal.sh"           "$CAP/selo.sh.txt"
[ -f "$A/adversarial-wp3-verdict.md" ] && cp "$A/adversarial-wp3-verdict.md" "$CAP/adversarial-wp3-verdict.md.txt"
[ -f "$A/zz-wp3-colhe-veredicto.py" ] && cp "$A/zz-wp3-colhe-veredicto.py" "$CAP/colhe-veredicto.py.txt"
echo "  arquivos em captures/: $(ls -1 "$CAP" | wc -l)"

echo "=== 2. armadilhas do .gitignore ==="
echo "  arquivos em raw/: $(find "$SELO" -path '*/raw/*' -type f | wc -l) (esperado 0)"
echo "  arquivos *.log (nao .log.txt): $(find "$SELO" -name '*.log' -type f | wc -l) (esperado 0)"
IGNORADOS=$(find "$SELO" -type f -print0 | xargs -0 git check-ignore 2>/dev/null | wc -l)
echo "  arquivos do selo ignorados pelo git: $IGNORADOS (esperado 0)"
[ "$IGNORADOS" -eq 0 ] || { echo "  !! FALHA: arquivo do selo seria ignorado"; FAIL=1; }

echo "=== 3. prettier nos .md do selo ==="
npx prettier --write "$SELO/README.md" "$SELO/SPEC.md" > /dev/null 2>&1
npx prettier --check "$SELO/README.md" "$SELO/SPEC.md" > "$A/wp3-prettier.log" 2>&1
PRETTIER=$?
echo "  PRETTIER_CHECK_EXIT=$PRETTIER"
[ "$PRETTIER" -eq 0 ] || { echo "  !! FALHA: prettier"; FAIL=1; }

echo "=== 4. manifesto (da raiz do repo) ==="
rm -f "$SELO/MANIFEST.sha256"
find "$SELO" -type f ! -name 'MANIFEST.sha256' -print0 | sort -z | xargs -0 sha256sum > "$SELO/MANIFEST.sha256"
FORA=$(grep -cvE "^[0-9a-f]{64}  $SELO/" "$SELO/MANIFEST.sha256" || true)
echo "  caminhos fora do selo no manifesto: $FORA (esperado 0)"
[ "$FORA" -eq 0 ] || { echo "  !! FALHA: manifesto com caminho fora do selo"; FAIL=1; }
DESCOBERTOS=$(find "$SELO" -type f ! -name 'MANIFEST.sha256' | wc -l)
LISTADOS=$(wc -l < "$SELO/MANIFEST.sha256")
echo "  descobertos=$DESCOBERTOS listados=$LISTADOS"
[ "$DESCOBERTOS" -eq "$LISTADOS" ] && echo "  checked === discovered: OK" || { echo "  !! FALHA: cardinalidade"; FAIL=1; }

echo "=== 5. verificacao do manifesto ==="
LC_ALL=C sha256sum -c "$SELO/MANIFEST.sha256" > "$A/wp3-manifest-check.txt" 2>&1
OK=$(grep -c ': OK$' "$A/wp3-manifest-check.txt" || true)
NAOOK=$(grep -c ': FAILED$' "$A/wp3-manifest-check.txt" || true)
echo "  OK=$OK NAO-OK=$NAOOK (exit=$?)"
[ "$NAOOK" -eq 0 ] || { echo "  !! FALHA: manifesto nao confere"; FAIL=1; }

echo "=== 6. vínculo commit <-> selo ==="
echo "  (feito depois do commit, com o script de vinculo)"
echo
echo "  tamanho do selo: $(du -sh "$SELO" | cut -f1)"
if [ "$FAIL" -eq 0 ]; then echo "SELO: OK"; echo "SEAL_SCRIPT_EXIT=0"; else echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; fi
