#!/usr/bin/env bash
# Falsificacao de FRONTEIRA do WP3 (F-D2-depth-pin).
#
# A pergunta que este script responde, e que nenhum teste responde sozinho:
# a suite NOVA distingue o limite real das DUAS direcoes — 3 e 5 — enquanto a
# ANTIGA nao distinguia nenhuma? Se a antiga tambem reprovar, o WP nao pinou
# nada; se so a descida for medida, metade do pin e leitura de codigo, nao prova.
#
# As duas metades:
#   descida  (limite 3): o caso depth=3 deve CAIR — o ultimo elo inspecionado
#                        deixou de ser inspecionado.
#   subida   (limite 5): o caso depth=4 deve CAIR — o primeiro elo fora do
#                        limite passou a ser mapeado, e o caso exige que seja
#                        relancado como veio.
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts
LOG=$A/wp3-falsifica.log
# A saida do vitest carrega ANSI quando o ambiente tem cor, e os greps de
# veredicto deste script procuram texto puro. Forcar aqui para que o veredicto
# nao dependa de quem executa (na 1a rodada o `NO_COLOR` ambiente era implicito).
export NO_COLOR=1
exec > >(tee "$LOG") 2>&1

IMPL=src/lib/products.functions.ts
TESTE=src/test/products-fk-conflict.test.ts
ANTIGO=src/test/zzz-antigo-depth.test.ts
ORIG=$(sha256sum "$IMPL" | cut -d' ' -f1)
echo "sha256 do impl ANTES: $ORIG"
echo "limite no arquivo:   $(grep -n '^const FK_CAUSE_CHAIN_LIMIT' "$IMPL")"

resumo() { grep -E 'Test Files|Tests +[0-9]' "$1" | sed 's/^/    /'; }
falhas() { grep -E '^\s*(×|FAIL)' "$1" | head -4 | sed 's/^/    /'; }
muta() {
  sed -i "s/^const FK_CAUSE_CHAIN_LIMIT = 4;/const FK_CAUSE_CHAIN_LIMIT = $1;/" "$IMPL"
  echo "  limite agora: $(grep -n '^const FK_CAUSE_CHAIN_LIMIT' "$IMPL")"
}
restaura() {
  sed -i "s/^const FK_CAUSE_CHAIN_LIMIT = $1;/const FK_CAUSE_CHAIN_LIMIT = 4;/" "$IMPL"
  echo "  limite restaurado: $(grep -n '^const FK_CAUSE_CHAIN_LIMIT' "$IMPL")"
}

echo
echo "=== 1. suite NOVA, limite real (4) — deve PASSAR ==="
node node_modules/vitest/vitest.mjs run "$TESTE" > $A/f1.txt 2>&1
echo "  EXIT=$?"
resumo $A/f1.txt

echo
echo "=== 2. METADE DE BAIXO: limite 3 — a suite NOVA deve REPROVAR no caso depth=3 ==="
muta 3
node node_modules/vitest/vitest.mjs run "$TESTE" > $A/f2.txt 2>&1
echo "  EXIT=$?"
resumo $A/f2.txt
echo "  casos que falharam:"
falhas $A/f2.txt
restaura 3

echo
echo "=== 3. suite ANTIGA (do pai), MESMO limite 3 — deve PASSAR ==="
git show HEAD:"$TESTE" > "$ANTIGO"
echo "  casos no arquivo antigo: $(grep -c '^  it(' "$ANTIGO")"
node node_modules/vitest/vitest.mjs run "$ANTIGO" > $A/f3.txt 2>&1
echo "  EXIT=$?"
resumo $A/f3.txt
rm -f "$ANTIGO"

echo
echo "=== 4. METADE DE CIMA: limite 5 — a suite NOVA deve REPROVAR no caso depth=4 ==="
muta 5
node node_modules/vitest/vitest.mjs run "$TESTE" > $A/f4.txt 2>&1
echo "  EXIT=$?"
resumo $A/f4.txt
echo "  casos que falharam:"
falhas $A/f4.txt
restaura 5

echo
echo "=== 5. a suite ANTIGA, MESMO limite 5 — deve PASSAR (nao distinguia) ==="
git show HEAD:"$TESTE" > "$ANTIGO"
node node_modules/vitest/vitest.mjs run "$ANTIGO" > $A/f5.txt 2>&1
echo "  EXIT=$?"
resumo $A/f5.txt
rm -f "$ANTIGO"

echo
echo "=== 6. restaura o limite e confere byte a byte ==="
NOVO=$(sha256sum "$IMPL" | cut -d' ' -f1)
echo "  sha256 do impl DEPOIS: $NOVO"
if [ "$ORIG" = "$NOVO" ]; then echo "  RESTAURO: OK (byte a byte)"; else echo "  RESTAURO: FALHOU"; fi

echo
echo "=== 7. nao sobrou sonda no git (ASSERTADO, nao so impresso) ==="
SOBRAS=$(git status --short | grep -cE 'zzz-antigo-depth|zzz-wp3' || true)
echo "  arquivos de sonda no git status: $SOBRAS (esperado 0)"
echo "  --- git status completo ---"
git status --short | sed 's/^/    /'

EXIT1=$(grep -cE 'Tests +11 passed' $A/f1.txt)
# IDENTIDADE, nao cardinalidade: a falha tem de estar NA LINHA de falha (`×`) do
# caso de fronteira, e tem de ser a UNICA. O nome aparece tambem em linhas `✓`
# do reporter, entao contar o nome em qualquer lugar nao prova qual caso caiu
# (S6 N1, fail-open). Ancora no `×` e exige queda unica.
FALHOU2=$(grep -cE 'Tests +.*failed' $A/f2.txt)
NOME2=$(grep -cF '× fronteira: um 23503 no último elo inspecionado (depth=3) ainda é encontrado' $A/f2.txt)
QUEDA2=$(grep -cE '^\s*× ' $A/f2.txt)
EXIT3=$(grep -cE 'Tests +9 passed' $A/f3.txt)
NAO_FALHOU3=$(grep -cE 'Tests +.*failed' $A/f3.txt)
FALHOU4=$(grep -cE 'Tests +.*failed' $A/f4.txt)
NOME4=$(grep -cF '× fronteira: um 23503 em depth=4 (o primeiro elo fora do limite) é relançado como veio' $A/f4.txt)
QUEDA4=$(grep -cE '^\s*× ' $A/f4.txt)
EXIT5=$(grep -cE 'Tests +9 passed' $A/f5.txt)
NAO_FALHOU5=$(grep -cE 'Tests +.*failed' $A/f5.txt)
echo
echo "  --- descida (limite 3) ---"
echo "  suite nova verde com limite 4:       $EXIT1 (esperado 1)"
  echo "  suite nova com falhas no limite 3:   $FALHOU2 (esperado >= 1)"
  echo "  casos que falharam no limite 3:      $QUEDA2 (esperado 1)"
  echo "  a falha e no caso de FRONTEIRA 3:    $NOME2 (esperado 1)"
echo "  suite antiga verde com limite 3:     $EXIT3 (esperado 1)"
echo "  suite antiga NAO falhou:             $NAO_FALHOU3 (esperado 0)"
echo "  --- subida (limite 5) ---"
  echo "  suite nova com falhas no limite 5:   $FALHOU4 (esperado >= 1)"
  echo "  casos que falharam no limite 5:      $QUEDA4 (esperado 1)"
  echo "  a falha e no caso de FRONTEIRA 4:    $NOME4 (esperado 1)"
echo "  suite antiga verde com limite 5:     $EXIT5 (esperado 1)"
echo "  suite antiga NAO falhou:             $NAO_FALHOU5 (esperado 0)"
echo "  --- integridade ---"
echo "  restauro byte a byte:                $([ "$ORIG" = "$NOVO" ] && echo OK || echo FALHOU)"
echo "  sondas no git:                       $SOBRAS (esperado 0)"

if [ "$EXIT1" -eq 1 ] && [ "$FALHOU2" -ge 1 ] && [ "$NOME2" -eq 1 ] && [ "$QUEDA2" -eq 1 ] && \
   [ "$EXIT3" -eq 1 ] && [ "$NAO_FALHOU3" -eq 0 ] && \
   [ "$FALHOU4" -ge 1 ] && [ "$NOME4" -eq 1 ] && [ "$QUEDA4" -eq 1 ] && \
   [ "$EXIT5" -eq 1 ] && [ "$NAO_FALHOU5" -eq 0 ] && \
   [ "$ORIG" = "$NOVO" ] && [ "$SOBRAS" -eq 0 ]; then
  echo "  VEREDITO: FRONTEIRA PINADA NAS DUAS DIRECOES - a nova distingue 3 e 5 de 4; a antiga nao distinguia nenhum"
  echo "FALSIFICA_EXIT=0"
else
  echo "  VEREDITO: INCONCLUSIVO - ver as saidas acima"
  echo "FALSIFICA_EXIT=1"
fi
