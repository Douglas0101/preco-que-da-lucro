def load_original_rows():
    """Lê a camada da MEDIÇÃO ORIGINAL a partir do commit — por PADRÃO, não por linha."""
    text = subprocess.run(
        ["git", "show", f"{ORIGIN_COMMIT}:{ANEXO_REL}"],
        cwd=ROOT,
        capture_output=True,
        text=True,
        check=True,
    ).stdout
    rows = []
    for line in text.splitlines():
        if ID_RE.match(line):
            rows.append(line)
    ids = [ID_RE.match(r).group(1) for r in rows]
    if len(ids) != len(set(ids)):
        dup = sorted({i for i in ids if ids.count(i) > 1})
        print(f"FALHOU: ids duplicados na origem: {dup}", file=sys.stderr)
        raise SystemExit(1)
    return rows, ids


def split_row(line: str):
    return [c.strip() for c in line.strip().strip("|").split("|")]
