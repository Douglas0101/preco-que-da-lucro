#!/usr/bin/env python3
"""Verificação independente da camada recomputada (identidade, não cardinalidade)."""
import re
import subprocess
from collections import Counter
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
CAMADA = ROOT / "docs/evidence/trilho-c-recompute-36-40-2026-09-19/camada-36-40-recomputada.md"
ANEXO_REL = "docs/evidence/plan-recap-2026-09-15/ANEXO-ITENS-2026-09-15.md"
ORIGIN = "d9e58a28d21c0178f2b810d00e38d4b4866e82d8"
ID_RE = re.compile(r"^\|\s*`((?:P0|P1|P2|ORD|GATE|DEF)-\d+)`")

linhas = [l for l in CAMADA.read_text(encoding="utf-8").splitlines() if l.lstrip().startswith("|")]
tabela = [l for l in linhas if ID_RE.match(l)]
nf = Counter(l.count("|") for l in tabela)
ids = [ID_RE.match(l).group(1) for l in tabela]

orig = subprocess.run(
    ["git", "show", f"{ORIGIN}:{ANEXO_REL}"], cwd=ROOT, capture_output=True, text=True, check=True
).stdout
orig_tabela = [l for l in orig.splitlines() if ID_RE.match(l)]
orig_ids = [ID_RE.match(l).group(1) for l in orig_tabela]

status = Counter(re.sub(r"[`*]", "", l.strip().strip("|").split("|")[1]).strip() for l in tabela)


def cells(line):
    return [c.strip() for c in line.strip().strip("|").split("|")]


mudadas = [
    ID_RE.match(a).group(1)
    for a, b in zip(tabela, orig_tabela)
    if cells(a) != cells(b)
]

print(f"linhas de tabela      = {len(tabela)}  (esperado 89)")
print(f"ids unicos            = {len(set(ids))}")
print(f"pipes por linha (NF)  = {dict(nf)}  (esperado {{6: 89}} -> 5 colunas + 2 bordas)")
print(f"status                = {dict(status)}")
print(f"ids identicos a origem= {ids == orig_ids}")
print(f"linhas com celulas diferentes da origem = {len(mudadas)}")
print(f"  itens: {mudadas}")

ok = (
    len(tabela) == 89
    and len(set(ids)) == 89
    and set(nf) == {6}
    and ids == orig_ids
    and status == {"DONE": 69, "PARTIAL": 10, "NS": 9, "NA": 1}
)
print(f"\nVERIFICACAO INDEPENDENTE: {'OK' if ok else 'FALHOU'}")
raise SystemExit(0 if ok else 1)
