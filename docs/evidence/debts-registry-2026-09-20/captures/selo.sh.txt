#!/usr/bin/env bash
# Selo do WP-R1.
set -u
cd "$(dirname "$0")/.." || exit 1
A=.artifacts; SELO=docs/evidence/debts-registry-2026-09-20; CAP=$SELO/captures; FAIL=0
echo "=== selo do WP-R1 ==="
for f in r1-red.log r1-s6red.log r1-s6green.log r1-gate.log r1-s7.log adversarial-r1-verdict.md; do
  [ -f "$A/$f" ] || { echo "  !! FALTA: $A/$f"; FAIL=1; }
done
[ "$FAIL" -eq 0 ] || { echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; exit 1; }
GENV=$(grep -c "CHECK_EXIT=0" "$A/r1-gate.log" || true)
[ "$GENV" -ge 1 ] || { echo "  !! FALHA: gate sem CHECK_EXIT=0"; exit 1; }
rm -rf "$CAP"; mkdir -p "$CAP"
cp "$A/r1-red.log" "$CAP/red-vitest.log.txt"
cp "$A/r1-s6red.log" "$CAP/s6-red-vitest.log.txt"
cp "$A/r1-s6green.log" "$CAP/s6-green-vitest.log.txt"
cp "$A/r1-gate.log" "$CAP/gate-local.log.txt"
cp "$A/r1-s7.log" "$CAP/s7-guard.log.txt"
cp "$A/adversarial-r1-verdict.md" "$CAP/adversarial-r1-verdict.md.txt"
cp scripts/m02-debts-guard.mjs "$CAP/guard.mjs.txt"
cp docs/evidence/agent-state/DEBTS.md "$CAP/debts.md.txt"
cp "$A/zz-r1-s7.sh" "$CAP/s7-guard.sh.txt"
cp "$A/zz-r1-seal.sh" "$CAP/selo.sh.txt"
: > "$CAP/hashes.txt"
for f in scripts/m02-debts-guard.mjs src/test/m02-debts-guard.test.ts \
         docs/evidence/agent-state/DEBTS.md docs/evidence/debts-registry-2026-09-20/SPEC.md \
         AGENTS.md package.json .github/workflows/ui-stack.yml .github/workflows/ci-light.yml; do
  if [ -f "$f" ]; then sha256sum "$f" >> "$CAP/hashes.txt"; else echo "  !! FALTA no hashes: $f"; FAIL=1; fi
done
npx prettier --write "$SELO/README.md" "$SELO/SPEC.md" >/dev/null 2>&1
npx prettier --check "$SELO/README.md" "$SELO/SPEC.md" > "$A/r1-prettier.log" 2>&1 || { echo "  !! FALHA: prettier"; FAIL=1; }
rm -f "$SELO/MANIFEST.sha256"
find "$SELO" -type f ! -name 'MANIFEST.sha256' -print0 | sort -z | xargs -0 sha256sum > "$SELO/MANIFEST.sha256"
D=$(find "$SELO" -type f ! -name 'MANIFEST.sha256' | wc -l); L=$(wc -l < "$SELO/MANIFEST.sha256")
echo "  descobertos=$D listados=$L"; [ "$D" -eq "$L" ] || { echo "  !! cardinalidade"; FAIL=1; }
LC_ALL=C sha256sum -c "$SELO/MANIFEST.sha256" > "$A/r1-manifest-check.txt" 2>&1
N=$(grep -c ': FAILED$' "$A/r1-manifest-check.txt" || true); echo "  NAO-OK=$N"; [ "$N" -eq 0 ] || FAIL=1
if [ "$FAIL" -eq 0 ]; then echo "SELO: OK"; echo "SEAL_SCRIPT_EXIT=0"; else echo "SELO: REPROVADO"; echo "SEAL_SCRIPT_EXIT=1"; fi
